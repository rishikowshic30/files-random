Query Business Rule - to prevent unauthorized access to the data



(function executeRule(current, previous /*null when async*/ ) {

    var current_user = gs.getUserID(); // -> used to get the name of the current logged in user

    var usr = new GlideRecord('sys_user');

    usr.addQuery('sys_id', current_user);

    usr.query();
    if (usr.next()) {
        current.addQuery('location=', usr.location);
    }

})(current, previous);
